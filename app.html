onAuthStateChanged(auth, async (user) => { if (user) { S.user = { uid: user.uid, email: user.email, displayName: user.displayName }; await load(user.uid); // If invite was just claimed by new signup — go directly to dashboard if (S._inviteClaimed) { delete S._inviteClaimed; await load(user.uid); go('dashboard'); // If user arrived via invite link and hasn't claimed yet — show accept-invite } else if (_inviteId && S.currentScreen === 'accept-invite') { go('accept-invite', { inviteId: _inviteId }); } else if (!S.firm) { // No firm profile — check if they're an invited staff member if (S.firmUsers?.length > 0) { // Has a firm_users record but firm profile failed to load — try dashboard go('dashboard'); } else { // Truly new user — start onboarding go('onboarding-firm'); } } else if (ONBOARDING_SCREENS.includes(S.currentScreen)) { // Returning user with currentScreen stuck on an onboarding screen go('setup'); } else if (!isSetupComplete() && S.currentScreen === 'dashboard') { // Returning user who hasn't finished setup — send to checklist go('setup'); } else { go(S.currentScreen || 'dashboard'); } } else { // Not authenticated if (_inviteId) { go('accept-invite', { inviteId: _inviteId }); } else { go('login'); } } });
